set(LM_TEST_ENV "")
if(LM_ENABLE_TSAN)
  # 系统包里的 gRPC / absl / protobuf 没有用 TSan 编译，其内部同步对 TSan 不可见，会误报
  # （如 absl GraphCycles::GetId）。这里只忽略“由这些库自身发起”的内存访问（called_from_lib），
  # 不用 race: 匹配栈帧，以免把我们自己代码里经由 gRPC 线程触发的真实竞争也抑制掉。
  # called_from_lib 的每条规则必须恰好匹配一个库，所以按实际存在的库逐个生成。
  set(_supp "${CMAKE_CURRENT_BINARY_DIR}/tsan.supp")
  file(GLOB _libs
    /lib/*/libabsl_*.so.* /usr/lib/*/libabsl_*.so.* /usr/lib/libabsl_*.so.*
    /lib/*/libgrpc*.so.* /usr/lib/*/libgrpc*.so.*
    /lib/*/libgpr.so.* /usr/lib/*/libgpr.so.*
    /lib/*/libprotobuf.so.* /usr/lib/*/libprotobuf.so.*
    /lib/*/libupb*.so.* /usr/lib/*/libupb*.so.*)
  set(_names "")
  foreach(_lib ${_libs})
    get_filename_component(_n ${_lib} NAME)
    string(REGEX REPLACE "\\.so\\..*$" ".so." _n ${_n})
    list(APPEND _names ${_n})
  endforeach()
  list(REMOVE_DUPLICATES _names)
  set(_content "# generated by tests/CMakeLists.txt\n")
  foreach(_n ${_names})
    string(APPEND _content "called_from_lib:${_n}\n")
  endforeach()
  # seqlock 读端按设计会与写端并发拷贝，撕裂结果靠 seq 比较丢弃（见 common/include/lm/seqlock_reader.h）
  string(APPEND _content "race:lm::seqlock_read\n")
  file(WRITE ${_supp} "${_content}")
  set(LM_TEST_ENV "TSAN_OPTIONS=halt_on_error=1 suppressions=${_supp}")
endif()

function(lm_add_test name)
  add_executable(${name} ${ARGN})
  target_link_libraries(${name} PRIVATE GTest::gtest_main)
  gtest_discover_tests(${name} DISCOVERY_TIMEOUT 30 PROPERTIES ENVIRONMENT "${LM_TEST_ENV}")
endfunction()
