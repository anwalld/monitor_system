# eBPF（M3）：Linux 上同时找到 clang、bpftool、libbpf 才启用；AUTO 缺任一项时跳过，ON 时报错
set(LM_HAVE_BPF OFF)
if(NOT LM_ENABLE_BPF STREQUAL "OFF")
  set(_why "")
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(_why "非 Linux")
  else()
    find_program(LM_BPF_CLANG NAMES clang clang-18 clang-17 clang-16 clang-15 clang-14)
    # Ubuntu 的 /usr/sbin/bpftool 是按 uname -r 转发的包装脚本，容器里会失败，优先用真实二进制
    file(GLOB _bpftools /usr/lib/linux-tools/*/bpftool)
    list(SORT _bpftools COMPARE NATURAL ORDER DESCENDING)
    if(BPFTOOL)
      set(LM_BPFTOOL ${BPFTOOL})
    elseif(_bpftools)
      list(GET _bpftools 0 LM_BPFTOOL)
    else()
      find_program(LM_BPFTOOL_PATH NAMES bpftool)
      set(LM_BPFTOOL ${LM_BPFTOOL_PATH})
    endif()
    find_package(PkgConfig QUIET)
    if(PkgConfig_FOUND)
      pkg_check_modules(LIBBPF QUIET IMPORTED_TARGET libbpf>=1.0)
    endif()
    if(NOT LM_BPF_CLANG)
      set(_why "找不到 clang")
    elseif(NOT LM_BPFTOOL)
      set(_why "找不到 bpftool")
    elseif(NOT LIBBPF_FOUND)
      set(_why "找不到 libbpf >= 1.0")
    elseif(NOT EXISTS ${LM_VMLINUX_BTF})
      set(_why "找不到 BTF ${LM_VMLINUX_BTF}")
    endif()
  endif()
  if(_why STREQUAL "")
    set(LM_HAVE_BPF ON)
    message(STATUS "eBPF：已启用（clang ${LM_BPF_CLANG}，bpftool ${LM_BPFTOOL}，libbpf ${LIBBPF_VERSION}）")
  elseif(LM_ENABLE_BPF STREQUAL "ON")
    message(FATAL_ERROR "eBPF：LM_ENABLE_BPF=ON 但无法启用：${_why}")
  else()
    message(STATUS "eBPF：未启用（${_why}）")
  endif()
else()
  message(STATUS "eBPF：未启用（LM_ENABLE_BPF=OFF）")
endif()
